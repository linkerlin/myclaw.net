**ProjectTypeDetector.cs**:
```csharp
public class ProjectTypeDetector : IWorkspaceDetector<ProjectTypeInfo>
{
    // 项目签名数据库
    private static readonly Dictionary<ProjectType, ProjectSignature> Signatures = new()
    {
        [ProjectType.React] = new(
            new[] { "package.json" },
            new[] { "react", "react-dom" },
            new[] { "src/App.tsx", "src/App.jsx", "src/App.js" }
        ),
        [ProjectType.Vue] = new(
            new[] { "package.json" },
            new[] { "vue" },
            new[] { "vue.config.js", "src/App.vue" }
        ),
        [ProjectType.DotNet] = new(
            new[] { ".csproj", ".sln" },
            Array.Empty<string>(),
            Array.Empty<string>()
        ),
        // ... 更多类型
    };

    public async Task<ProjectTypeInfo> DetectAsync(string path, int depth = 2)
    {
        var scores = new Dictionary<ProjectType, double>();
        
        foreach (var (type, signature) in Signatures)
        {
            var score = await CalculateScoreAsync(path, signature, depth);
            if (score > 0.3) // 最小置信度
            {
                scores[type] = score;
            }
        }
        
        return scores.Count > 0 
            ? new ProjectTypeInfo(scores.OrderByDescending(s => s.Value).First())
            : ProjectTypeInfo.Unknown;
    }
    
    private async Task<double> CalculateScoreAsync(
        string path, 
        ProjectSignature signature, 
        int depth)
    {
        double score = 0;
        int checks = 0;
        
        // 检查必需文件
        foreach (var file in signature.RequiredFiles)
        {
            if (await FileExistsAsync(path, file, depth))
            {
                score += 0.4;
            }
            checks++;
        }
        
        // 检查依赖 (package.json 等)
        foreach (var dep in signature.Dependencies)
        {
            if (await HasDependencyAsync(path, dep))
            {
                score += 0.3;
            }
            checks++;
        }
        
        // 检查可选文件
        foreach (var file in signature.OptionalFiles)
        {
            if (await FileExistsAsync(path, file, depth))
            {
                score += 0.15;
            }
            checks++;
        }
        
        return checks > 0 ? score / checks : 0;
    }
}
```

**ContextLineBuilder.cs**:

```csharp
public class ContextLineBuilder
{
    public async Task<string> BuildContextLineAsync(string workspacePath)
    {
        var parts = new List<string>();
        
        // 项目信息
        var projectInfo = await _projectDetector.DetectAsync(workspacePath);
        if (projectInfo.Type != ProjectType.Unknown)
        {
            parts.Add($"Project: {projectInfo.Name ?? projectInfo.Type.ToString()}");
        }
        
        // 路径
        parts.Add($"Path: {workspacePath}");
        
        // Git 信息
        var gitInfo = await _gitDetector.DetectAsync(workspacePath);
        if (gitInfo.IsGitRepo)
        {
            var gitLine = $"Git: {gitInfo.Branch}";
            if (gitInfo.IsDirty)
            {
                gitLine += $" | dirty (+{gitInfo.ChangedFiles.Count} files)";
            }
            parts.Add(gitLine);
        }
        
        // 技术栈
        var techStack = await _techStackDetector.DetectAsync(workspacePath);
        if (techStack.Technologies.Any())
        {
            parts.Add($"Stack: {string.Join(", ", techStack.Technologies.Take(5))}");
        }
        
        return string.Join(" | ", parts);
    }
}
```

```csharp
public class AnalyticsTracker : IAnalyticsTracker
{
    private readonly AnalyticsStorage _storage;
    private readonly string _sessionId;
    private readonly DateTime _sessionStart;
    
    public async Task TrackToolCallAsync(string toolName, ToolCallContext context)
    {
        var record = new ToolCallRecord
        {
            Id = Guid.NewGuid().ToString(),
            SessionId = _sessionId,
            ToolName = toolName,
            Timestamp = DateTime.UtcNow,
            Success = context.Success,
            DurationMs = context.DurationMs,
            ErrorMessage = context.Error?.Message,
            WorkspacePath = context.WorkspacePath,
            Metadata = context.Metadata
        };
        
        await _storage.SaveAsync(record);
    }
    
    public async Task<UsageStatistics> GetStatisticsAsync(TimeSpan period)
    {
        var endTime = DateTime.UtcNow;
        var startTime = endTime - period;
        
        var records = await _storage.QueryAsync(startTime, endTime);
        
        return new UsageStatistics
        {
            Period = period,
            TotalToolCalls = records.Count,
            UniqueTools = records.Select(r => r.ToolName).Distinct().Count(),
            ToolBreakdown = records
                .GroupBy(r => r.ToolName)
                .Select(g => new ToolStatistics
                {
                    ToolName = g.Key,
                    CallCount = g.Count(),
                    SuccessCount = g.Count(r => r.Success),
                    FailureCount = g.Count(r => !r.Success),
                    AverageDuration = g.Average(r => r.DurationMs),
                    LastUsed = g.Max(r => r.Timestamp)
                })
                .OrderByDescending(s => s.CallCount)
                .ToList(),
            DailyBreakdown = records
                .GroupBy(r => r.Timestamp.Date)
                .Select(g => new DailyStatistics
                {
                    Date = g.Key,
                    CallCount = g.Count(),
                    UniqueTools = g.Select(r => r.ToolName).Distinct().Count()
                })
                .OrderBy(d => d.Date)
                .ToList()
        };
    }
}
```

```csharp
public class DailyBriefingGenerator
{
    public async Task<DailyBriefing> GenerateAsync(string workspacePath)
    {
        var yesterday = DateTime.UtcNow.Date.AddDays(-1);
        
        // 1. 获取昨日统计
        var stats = await _analytics.GetStatisticsAsync(TimeSpan.FromDays(1));
        
        // 2. 获取新增记忆
        var newMemories = await _memoryStore.GetMemoriesSinceAsync(yesterday);
        
        // 3. 获取新增实体
        var newEntities = await _entityStore.GetEntitiesSinceAsync(yesterday);
        
        // 4. 生成简报
        var briefing = new DailyBriefing
        {
            Date = yesterday,
            GeneratedAt = DateTime.UtcNow,
            
            ToolUsage = new ToolUsageSummary
            {
                TotalCalls = stats.TotalToolCalls,
                TopTools = stats.ToolBreakdown.Take(5).ToList()
            },
            
            MemorySummary = new MemorySummary
            {
                NewMemories = newMemories.Count,
                NewEntities = newEntities.Count,
                KeyTopics = ExtractKeyTopics(newMemories)
            },
            
            Suggestions = await GenerateSuggestionsAsync(stats, newMemories),
            
            ContinuityItems = await ExtractContinuityItemsAsync(workspacePath)
        };
        
        return briefing;
    }
    
    private async Task<List<string>> GenerateSuggestionsAsync(
        UsageStatistics stats, 
        List<Memory> memories)
    {
        var suggestions = new List<string>();
        
        // 检测重复工具使用
        var repetitiveTools = stats.ToolBreakdown
            .Where(t => t.CallCount > 5)
            .Select(t => t.ToolName)
            .ToList();
        
        if (repetitiveTools.Any())
        {
            suggestions.Add($"你经常使用 {string.Join(", ", repetitiveTools)}，建议创建 Skill 自动化");
        }
        
        // 检测新概念
        var newConcepts = memories
            .Where(m => m.Type == MemoryType.Concept)
            .Take(3)
            .Select(m => m.Title)
            .ToList();
        
        if (newConcepts.Any())
        {
            suggestions.Add($"最近讨论了 {string.Join(", ", newConcepts)}，是否需要深入了解？");
        }
        
        return suggestions;
    }
}
```

### Phase 3: 进化算法增强

#### 语义模式检测

```csharp
public class SemanticPatternDetector
{
    // 使用简单的 TF-IDF + 余弦相似度进行语义聚类
    public async Task<List<SemanticCluster>> ClusterAsync(List<MemoryLog> logs)
    {
        // 1. 提取词频向量
        var documents = logs.Select(l => Tokenize(l.Content)).ToList();
        var vocabulary = BuildVocabulary(documents);
        var vectors = documents.Select(d => Vectorize(d, vocabulary)).ToList();
        
        // 2. K-means 聚类
        var clusters = KMeansCluster(vectors, k: Math.Min(5, logs.Count / 3));
        
        // 3. 为每个聚类提取主题
        return clusters.Select((c, i) => new SemanticCluster
        {
            Id = i,
            Logs = c.Indices.Select(idx => logs[idx]).ToList(),
            Centroid = c.Centroid,
            Theme = ExtractTheme(c, vocabulary),
            Confidence = CalculateConfidence(c)
        }).ToList();
    }
    
    private string ExtractTheme(Cluster cluster, Vocabulary vocab)
    {
        // 提取中心向量中权重最高的词
        var topWords = cluster.Centroid
            .Select((weight, idx) => (word: vocab.GetWord(idx), weight))
            .OrderByDescending(x => x.weight)
            .Take(5)
            .Select(x => x.word)
            .ToList();
        
        return string.Join(", ", topWords);
    }
}
```

### 方案 B: 安装脚本

**install.sh**:
```bash
#!/bin/bash
set -e

REPO="your-repo/myclaw.net"
VERSION="${1:-latest}"
INSTALL_DIR="${HOME}/.local/bin"

# 检测平台
OS=$(uname -s)
ARCH=$(uname -m)

case "$OS" in
    Linux*)     PLATFORM="linux";;
    Darwin*)    PLATFORM="osx";;
    CYGWIN*|MINGW*|MSYS*) PLATFORM="win";;
    *)          echo "不支持的操作系统: $OS"; exit 1;;
esac

case "$ARCH" in
    x86_64)     ARCH="x64";;
    arm64|aarch64) ARCH="arm64";;
    *)          echo "不支持的架构: $ARCH"; exit 1;;
esac

# 下载
echo "下载 myclaw $VERSION for $PLATFORM-$ARCH..."
URL="https://github.com/$REPO/releases/download/$VERSION/myclaw-$PLATFORM-$ARCH"
mkdir -p "$INSTALL_DIR"
curl -L "$URL" -o "$INSTALL_DIR/myclaw"
chmod +x "$INSTALL_DIR/myclaw"

echo "安装完成! 请确保 $INSTALL_DIR 在你的 PATH 中"
```

**使用**:
```bash
curl -sSL https://raw.githubusercontent.com/your-repo/myclaw.net/main/install.sh | bash
```
